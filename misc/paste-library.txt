Usage

#show all options
perl bin/insult.pl --help

#one insult password (adjective + adjective + noun + 4 digits)
perl bin/insult.pl

#five insult passwords, each ending with a symbol
perl bin/insult.pl -S -n 5

#stronger: four English words of 4-8 letters + 4 digits + symbol
perl bin/insult.pl -s english -S

#save ten passwords to a file (strength line is not included)
perl bin/insult.pl -n 10 >tmp/passwords.txt
